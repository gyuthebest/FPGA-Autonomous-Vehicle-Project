// 추가함: AXI Read Register Offset 정의
#define READ_REG0_OFFSET   0x00
#define READ_REG1_OFFSET   0x04
#define READ_REG2_OFFSET   0x08
#define READ_REG3_OFFSET   0x0C
#define READ_REG4_OFFSET   0x10
#define READ_REG5_OFFSET   0x14
#define READ_REG6_OFFSET   0x18
#define READ_REG7_OFFSET   0x1C
#define READ_REG8_OFFSET   0x20
#define READ_REG9_OFFSET   0x24
#define READ_REG10_OFFSET  0x28
#define READ_REG11_OFFSET  0x2C
#define READ_REG12_OFFSET  0x30
// 추가함: UDP 송신용 포트 및 IP 주소 정의
#define CARLA_UDP_PORT  5002        // send_output_to_carla()에서 사용하기 위해 carla ip를 전역 변수로 선언함

static ip_addr_t carla_ip;

struct udp_pcb *carla_tx_pcb = NULL;

// 추가함: UDP packet 수신 시 호출되는 콜백 함수
// 수신된 packet 길이 출력 후 buffer 해제
void carla_udp_receive_callback(
    void *arg,
    struct udp_pcb *pcb,
    struct pbuf *p,
    const ip_addr_t *addr,
    u16_t port)
{
    if (p == NULL) {
        return;
    }

    if (p->tot_len != 40) {
        xil_printf(
            "Invalid CARLA packet length: %d\r\n",
            p->tot_len
        );

        pbuf_free(p);
        return;
    }

    // UDP packet에서 reg0~reg9 복원
    uint32_t reg0;
    uint32_t reg1;
    uint32_t reg2;
    uint32_t reg3;
    uint32_t reg4;
    uint32_t reg5;
    uint32_t reg6;
    uint32_t reg7;
    uint32_t reg8;
    uint32_t reg9;

    pbuf_copy_partial(p, &reg0, 4, 0);
    pbuf_copy_partial(p, &reg1, 4, 4);
    pbuf_copy_partial(p, &reg2, 4, 8);
    pbuf_copy_partial(p, &reg3, 4, 12);
    pbuf_copy_partial(p, &reg4, 4, 16);
    pbuf_copy_partial(p, &reg5, 4, 20);
    pbuf_copy_partial(p, &reg6, 4, 24);
    pbuf_copy_partial(p, &reg7, 4, 28);
    pbuf_copy_partial(p, &reg8, 4, 32);
    pbuf_copy_partial(p, &reg9, 4, 36);

    // reg0~reg9를 AXI 레지스터에 write
    Xil_Out32(SENSOR_IP_BASEADDR + REG0_OFFSET, reg0);
    Xil_Out32(SENSOR_IP_BASEADDR + REG1_OFFSET, reg1);
    Xil_Out32(SENSOR_IP_BASEADDR + REG2_OFFSET, reg2);
    Xil_Out32(SENSOR_IP_BASEADDR + REG3_OFFSET, reg3);
    Xil_Out32(SENSOR_IP_BASEADDR + REG4_OFFSET, reg4);
    Xil_Out32(SENSOR_IP_BASEADDR + REG5_OFFSET, reg5);
    Xil_Out32(SENSOR_IP_BASEADDR + REG6_OFFSET, reg6);
    Xil_Out32(SENSOR_IP_BASEADDR + REG7_OFFSET, reg7);
    Xil_Out32(SENSOR_IP_BASEADDR + REG8_OFFSET, reg8);
    Xil_Out32(SENSOR_IP_BASEADDR + REG9_OFFSET, reg9);

    // packet buffer 해제
    pbuf_free(p);
}

// 추가함: UDP 수신 초기화 함수
int init_carla_udp_receiver(void)
{
    carla_rx_pcb = udp_new();                                       // UDP 통신 창구 생성

    if (carla_rx_pcb == NULL) {
        xil_printf("UDP PCB creation failed\r\n");
        return -1;
    }

    if (udp_bind(carla_rx_pcb, IP_ADDR_ANY, 5001) != ERR_OK) {      // PORT 연결
        xil_printf("UDP bind failed\r\n");
        return -1;
    }

    udp_recv(
        carla_rx_pcb,
        carla_udp_receive_callback,
        NULL
    );

    xil_printf("CARLA UDP receiver started on port 5001\r\n");

    return 0;
}

// 추가함: UDP 송신 초기화 함수
int init_carla_udp_sender(void)
{
    // UDP PCB 생성
    carla_tx_pcb = udp_new();

    if (carla_tx_pcb == NULL) {
        xil_printf("UDP TX PCB creation failed\r\n");
        return -1;
    }

    // CARLA PC의 IP 주소
    IP4_ADDR(
        &carla_ip,
        192, 168, 1, 20             // TODO: 실제 CARLA PC의 IP 주소로 변경 필요
    );

    xil_printf(
        "CARLA UDP sender started: %d\r\n",
        CARLA_UDP_PORT
    );

    return 0;
}

int send_output_to_carla(void)
{
    uint32_t read_reg0;
    uint32_t read_reg1;
    uint32_t read_reg2;
    uint32_t read_reg3;
    uint32_t read_reg4;
    uint32_t read_reg5;
    uint32_t read_reg6;
    uint32_t read_reg7;
    uint32_t read_reg8;
    uint32_t read_reg9;
    uint32_t read_reg10;
    uint32_t read_reg11;
    uint32_t read_reg12;

    struct pbuf *p;
    err_t err;

    if (carla_tx_pcb == NULL) {
        return -1;
    }

    // AXI read
    read_reg0  = Xil_In32(SENSOR_IP_BASEADDR + READ_REG0_OFFSET);
    read_reg1  = Xil_In32(SENSOR_IP_BASEADDR + READ_REG1_OFFSET);
    read_reg2  = Xil_In32(SENSOR_IP_BASEADDR + READ_REG2_OFFSET);
    read_reg3  = Xil_In32(SENSOR_IP_BASEADDR + READ_REG3_OFFSET);
    read_reg4  = Xil_In32(SENSOR_IP_BASEADDR + READ_REG4_OFFSET);
    read_reg5  = Xil_In32(SENSOR_IP_BASEADDR + READ_REG5_OFFSET);
    read_reg6  = Xil_In32(SENSOR_IP_BASEADDR + READ_REG6_OFFSET);
    read_reg7  = Xil_In32(SENSOR_IP_BASEADDR + READ_REG7_OFFSET);
    read_reg8  = Xil_In32(SENSOR_IP_BASEADDR + READ_REG8_OFFSET);
    read_reg9  = Xil_In32(SENSOR_IP_BASEADDR + READ_REG9_OFFSET);
    read_reg10 = Xil_In32(SENSOR_IP_BASEADDR + READ_REG10_OFFSET);
    read_reg11 = Xil_In32(SENSOR_IP_BASEADDR + READ_REG11_OFFSET);
    read_reg12 = Xil_In32(SENSOR_IP_BASEADDR + READ_REG12_OFFSET);

    // 52-byte UDP packet
    p = pbuf_alloc(
        PBUF_TRANSPORT,
        52,
        PBUF_RAM
    );

    if (p == NULL) {
        xil_printf("UDP TX pbuf allocation failed\r\n");
        return -1;
    }

    // register → packet
    memcpy((uint8_t *)p->payload +  0, &read_reg0,  4);
    memcpy((uint8_t *)p->payload +  4, &read_reg1,  4);
    memcpy((uint8_t *)p->payload +  8, &read_reg2,  4);
    memcpy((uint8_t *)p->payload + 12, &read_reg3,  4);
    memcpy((uint8_t *)p->payload + 16, &read_reg4,  4);
    memcpy((uint8_t *)p->payload + 20, &read_reg5,  4);
    memcpy((uint8_t *)p->payload + 24, &read_reg6,  4);
    memcpy((uint8_t *)p->payload + 28, &read_reg7,  4);
    memcpy((uint8_t *)p->payload + 32, &read_reg8,  4);
    memcpy((uint8_t *)p->payload + 36, &read_reg9,  4);
    memcpy((uint8_t *)p->payload + 40, &read_reg10, 4);
    memcpy((uint8_t *)p->payload + 44, &read_reg11, 4);
    memcpy((uint8_t *)p->payload + 48, &read_reg12, 4);

    // CARLA로 전송
    err = udp_sendto(
        carla_tx_pcb,
        p,
        &carla_ip,
        CARLA_UDP_PORT
    );

    pbuf_free(p);

    if (err != ERR_OK) {
        xil_printf(
            "UDP TX failed: %d\r\n",
            err
        );
        return -1;
    }

    return 0;
}

#if LWIP_IPV6==1
void print_ip6(char *msg, ip_addr_t *ip)
{
	print(msg);
	xil_printf(" %x:%x:%x:%x:%x:%x:%x:%x\n\r",
			IP6_ADDR_BLOCK1(&ip->u_addr.ip6),
			IP6_ADDR_BLOCK2(&ip->u_addr.ip6),
			IP6_ADDR_BLOCK3(&ip->u_addr.ip6),
			IP6_ADDR_BLOCK4(&ip->u_addr.ip6),
			IP6_ADDR_BLOCK5(&ip->u_addr.ip6),
			IP6_ADDR_BLOCK6(&ip->u_addr.ip6),
			IP6_ADDR_BLOCK7(&ip->u_addr.ip6),
			IP6_ADDR_BLOCK8(&ip->u_addr.ip6));

}
#else
void
print_ip(char *msg, ip_addr_t *ip)
{
	print(msg);
	xil_printf("%d.%d.%d.%d\n\r", ip4_addr1(ip), ip4_addr2(ip),
			ip4_addr3(ip), ip4_addr4(ip));
}

void
print_ip_settings(ip_addr_t *ip, ip_addr_t *mask, ip_addr_t *gw)
{

	print_ip("Board IP: ", ip);
	print_ip("Netmask : ", mask);
	print_ip("Gateway : ", gw);
}

int main()
{
#if LWIP_IPV6==0
	ip_addr_t ipaddr, netmask, gw;

#endif
	/* the mac address of the board. this should be unique per board */
	unsigned char mac_ethernet_address[] =
	{ 0x00, 0x0a, 0x35, 0x00, 0x01, 0x02 };

	echo_netif = &server_netif;
#if defined (__arm__) && !defined (ARMR5)
#if XPAR_GIGE_PCS_PMA_SGMII_CORE_PRESENT == 1 || XPAR_GIGE_PCS_PMA_1000BASEX_CORE_PRESENT == 1
	ProgramSi5324();
	ProgramSfpPhy();
#endif
#endif

/* Define this board specific macro in order perform PHY reset on ZCU102 */
#ifdef XPS_BOARD_ZCU102
	if(IicPhyReset()) {
		xil_printf("Error performing PHY reset \n\r");
		return -1;
	}
#endif

	init_platform();

#if LWIP_IPV6==0
#if LWIP_DHCP==1
    ipaddr.addr = 0;
	gw.addr = 0;
	netmask.addr = 0;
#else
	/* initialize IP addresses to be used */
	IP4_ADDR(&ipaddr,  192, 168,   1, 10);
	IP4_ADDR(&netmask, 255, 255, 255,  0);
	IP4_ADDR(&gw,      192, 168,   1,  1);
#endif
#endif
	print_app_header();

	lwip_init();

#if (LWIP_IPV6 == 0)
	/* Add network interface to the netif_list, and set it as default */
	if (!xemac_add(echo_netif, &ipaddr, &netmask,
						&gw, mac_ethernet_address,
						PLATFORM_EMAC_BASEADDR)) {
		xil_printf("Error adding N/W interface\n\r");
		return -1;
	}
#else
	/* Add network interface to the netif_list, and set it as default */
	if (!xemac_add(echo_netif, NULL, NULL, NULL, mac_ethernet_address,
						PLATFORM_EMAC_BASEADDR)) {
		xil_printf("Error adding N/W interface\n\r");
		return -1;
	}
	echo_netif->ip6_autoconfig_enabled = 1;

	netif_create_ip6_linklocal_address(echo_netif, 1);
	netif_ip6_addr_set_state(echo_netif, 0, IP6_ADDR_VALID);

	print_ip6("\n\rBoard IPv6 address ", &echo_netif->ip6_addr[0].u_addr.ip6);

#endif
	netif_set_default(echo_netif);

	/* now enable interrupts */
	platform_enable_interrupts();

	/* specify that the network if is up */
	netif_set_up(echo_netif);

#if (LWIP_IPV6 == 0)
#if (LWIP_DHCP==1)
	/* Create a new DHCP client for this interface.
	 * Note: you must call dhcp_fine_tmr() and dhcp_coarse_tmr() at
	 * the predefined regular intervals after starting the client.
	 */
	dhcp_start(echo_netif);
	dhcp_timoutcntr = 24;

	while(((echo_netif->ip_addr.addr) == 0) && (dhcp_timoutcntr > 0))
		xemacif_input(echo_netif);

	if (dhcp_timoutcntr <= 0) {
		if ((echo_netif->ip_addr.addr) == 0) {
			xil_printf("DHCP Timeout\r\n");
			xil_printf("Configuring default IP of 192.168.1.10\r\n");
			IP4_ADDR(&(echo_netif->ip_addr),  192, 168,   1, 10);
			IP4_ADDR(&(echo_netif->netmask), 255, 255, 255,  0);
			IP4_ADDR(&(echo_netif->gw),      192, 168,   1,  1);
		}
	}

	ipaddr.addr = echo_netif->ip_addr.addr;
	gw.addr = echo_netif->gw.addr;
	netmask.addr = echo_netif->netmask.addr;
#endif

	print_ip_settings(&ipaddr, &netmask, &gw);

#endif
	/* start the application (web server, rxtest, txtest, etc..) */
	start_application();

    // 추가함: UDP 수신 초기화
    if (init_carla_udp_receiver() != 0) {
        xil_printf("Failed to initialize CARLA UDP receiver\r\n");
        return -1;
    }

    // 추가함: UDP 송신 초기화
    if (init_carla_udp_sender() != 0) {
        xil_printf("Failed to initialize CARLA UDP sender\r\n");
        return -1;
    }

	/* receive and process packets */
	while (1) {
		if (TcpFastTmrFlag) {
			tcp_fasttmr();
			TcpFastTmrFlag = 0;
		}
		if (TcpSlowTmrFlag) {
			tcp_slowtmr();
			TcpSlowTmrFlag = 0;
		}
		xemacif_input(echo_netif);
		send_output_to_carla();
	}

	/* never reached */
	cleanup_platform();

	return 0;
}
